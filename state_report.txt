@app.before_request
def load_current_user():
    """Sa bawat request: alamin kung sino ang naka-login at kung valid pa ang session niya."""
    g.user = None
    if request.endpoint == 'static':
        return
    user_id = session.get('user_id')
    if user_id is None:
        return

    now = int(time.time())
    limit = REMEMBER_SECONDS if session.get('remember') else IDLE_SECONDS
    if now - session.get('last_seen', now) > limit:      # idle timeout
        session.clear()
        flash('Your session expired. Please log in again.', 'warning')
        return

    user = User.find(user_id)
    if user is None or not user.is_active or user.session_version != session.get('session_version'):
        session.clear()
        return
    session['last_seen'] = now
    g.user = user

STUDENT_TYPES = ('New', 'Transferee', 'Continuing', 'Returnee')


class Stats:
    """Mga bilang para sa mga dashboard."""

    @staticmethod
    def enrolled_by_type(term_id):
        """Ilang student ang Enrolled na sa term na ito, ayon sa uri nila."""
        rows = query("SELECT student_type, COUNT(*) AS n FROM enrollments "
                     "WHERE term_id = %s AND status = 'Enrolled' GROUP BY student_type", (term_id,))
        counts = {r['student_type']: r['n'] for r in rows}
        return {t: counts.get(t, 0) for t in STUDENT_TYPES}

    @classmethod
    def registrar(cls, term_id=None):
        """Lahat ng kailangan ng dashboard ng Registrar. Ang bukas na term ang default."""
        terms = Term.active()
        term = next((t for t in terms if t.id == term_id), terms[0] if terms else None)
        current = Term.current()
        by_type = cls.enrolled_by_type(current.id) if current else {t: 0 for t in STUDENT_TYPES}
        data = dict(terms=terms, term=term, counts={}, drafts=0, total=0, programs=[], types={},
                    current_term=current, by_type=by_type, enrolled_total=sum(by_type.values()))
        if term:
            rows = query("SELECT status, COUNT(*) AS n FROM applications WHERE term_id = %s GROUP BY status",
                         (term.id,))
            counts = {r['status']: r['n'] for r in rows}
            data['drafts'] = counts.pop('Draft', 0)
            data['counts'] = counts
            data['total'] = sum(counts.values())
            data['programs'] = query(
                "SELECT p.code, p.name, COUNT(*) AS n FROM applications a JOIN programs p ON p.id = a.program_id "
                "WHERE a.term_id = %s AND a.status <> 'Draft' GROUP BY p.id ORDER BY n DESC, p.code", (term.id,))
            type_rows = query("SELECT applicant_type, COUNT(*) AS n FROM applications "
                              "WHERE term_id = %s AND status <> 'Draft' GROUP BY applicant_type", (term.id,))
            data['types'] = {r['applicant_type']: r['n'] for r in type_rows}
        return data

import re
from datetime import date
from decimal import Decimal
from db import query, execute
from models.application import Application

class Quotation:
    """Ang tuition quotation ng Finance, kasama ang manual na Student ID. Ang total ay kinukuwenta ng system."""

    # (column, label). Sundan ang Accounting slip ng SIT.
    FEES = (
        ('tuition_fee', 'Tuition Fee'),
        ('energy_fee', 'Energy Fee'),
        ('book_fee', 'Book / Manual'),
        ('printouts_fee', 'Printouts'),
        ('admission_fee', 'Admission'),
    )
    STUDENT_ID_RE = re.compile(r'^\d{2}-[1-3]-\d{4}$')          # YY-T-NNNN, hal. 26-2-0110
    ACCOUNT_RE = re.compile(r'^[A-Za-z0-9-]{3,20}$')
    MONEY_RE = re.compile(r'^\d{1,7}(\.\d{1,2})?$')
    UNITS_RE = re.compile(r'^\d{1,2}(\.\d)?$')

    @staticmethod
    def find(application_id):
        return query("SELECT * FROM quotations WHERE application_id = %s", (application_id,), one=True)

    @staticmethod
    def application_for_finance(application_id):
        """Ang application kung puwede pang i-quote: For Enrollment, may Originals received, hindi archived ang term."""
        application = Application.find(application_id)
        if application is None or application.status != 'For Enrollment' or not application.originals_received_at:
            return None
        if application.term_id:
            term = query("SELECT admission_status FROM academic_terms WHERE id = %s", (application.term_id,), one=True)
            if term and term['admission_status'] == 'Archived':
                return None
        return application

    @staticmethod
    def suggest_account_no():
        n = query("SELECT COALESCE(MAX(id), 0) + 1 AS n FROM quotations", (), one=True)['n']
        return f'AC-{date.today().year}-{n:04d}'

    @staticmethod
    def to_form(row):
        """Ang naka-save na quotation bilang text, para sa form."""
        form = {'student_id': row['student_id'], 'account_no': row['account_no'],
                'units': format(row['units'].normalize(), 'f')}
        for name, _label in Quotation.FEES:
            form[name] = '{:.2f}'.format(row[name])
        return form

    @classmethod
    def clean(cls, form, application_id):
        """Sinusuri ang form. Ibinabalik ang (data, errors). Ang total ay kinukuwenta dito, hindi galing sa form."""
        data, errors = {}, []

        student_id = form.get('student_id', '').strip()
        if not cls.STUDENT_ID_RE.match(student_id):
            errors.append('Student ID must look like 26-2-0110 (year, trimester 1 to 3, number).')
        elif query("SELECT id FROM users WHERE username = %s", (student_id,), one=True) or \
                query("SELECT id FROM quotations WHERE student_id = %s AND application_id <> %s",
                      (student_id, application_id), one=True):
            errors.append(f'Student ID {student_id} is already taken.')
        data['student_id'] = student_id

        account_no = form.get('account_no', '').strip().upper()
        if not cls.ACCOUNT_RE.match(account_no):
            errors.append('Account No. may only contain letters, numbers and hyphens (3 to 20 characters).')
        elif query("SELECT id FROM quotations WHERE account_no = %s AND application_id <> %s",
                   (account_no, application_id), one=True):
            errors.append(f'Account No. {account_no} is already used.')
        data['account_no'] = account_no

        raw = form.get('units', '').strip()
        if cls.UNITS_RE.match(raw) and Decimal(raw) > 0:
            data['units'] = Decimal(raw)
        else:
            errors.append('Enter the number of units (greater than 0).')
            data['units'] = Decimal('0')

        total = Decimal('0')
        for name, label in cls.FEES:
            raw = form.get(name, '').strip().replace(',', '')
            if raw == '':
                value = Decimal('0')
            elif cls.MONEY_RE.match(raw):
                value = Decimal(raw)
            else:
                errors.append(f'{label} must be an amount like 1500 or 1500.50.')
                value = Decimal('0')
            data[name] = value
            total += value
        if data['tuition_fee'] <= 0 and not any(label.startswith('Tuition') for label in
                                                 [e for e in errors if e.startswith('Tuition')]):
            errors.append('Enter the Tuition Fee.')
        data['total'] = total
        return data, errors

    @classmethod
    def save(cls, application, data, by_user):
        """Ini-insert o ina-update ang quotation. Ibinabalik ang True kung bago."""
        if not application.term_id:
            raise ValueError('This application has no term yet.')
        existing = cls.find(application.id)
        fee_cols = [name for name, _label in cls.FEES]
        values = [data['student_id'], data['account_no'], data['units']] + [data[c] for c in fee_cols] + [data['total']]

        if existing:
            sets = ', '.join(f'{c} = %s' for c in ['student_id', 'account_no', 'units'] + fee_cols + ['total'])
            execute(f"UPDATE quotations SET {sets} WHERE application_id = %s", values + [application.id])
        else:
            cols = ', '.join(['application_id', 'student_id', 'account_no', 'units'] + fee_cols +
                             ['total', 'created_by'])
            marks = ', '.join(['%s'] * (len(fee_cols) + 6))
            execute(f"INSERT INTO quotations ({cols}) VALUES ({marks})",
                    [application.id] + values + [by_user.id])

        # Ang enrollment record: Quoted na. (Hindi ibababa kung Enrolled na.)
        execute("INSERT INTO enrollments (user_id, term_id, student_type, status) VALUES (%s, %s, %s, 'Quoted') "
                "ON DUPLICATE KEY UPDATE student_type = VALUES(student_type), "
                "status = IF(status = 'Enrolled', status, 'Quoted')",
                (application.user_id, application.term_id, application.applicant_type))

        owner = User.find(application.user_id)
        details = [('Student ID', data['student_id']), ('Account No.', data['account_no']),
                   ('Total', 'PHP {:,.2f}'.format(data['total']))]
        if existing:
            Notification.notify(owner, 'Your SIT quotation was updated', 'Quotation updated',
                                'The Finance office updated your tuition quotation. Please see the new total below.',
                                details)
        else:
            Notification.notify(owner, 'Your SIT tuition quotation', 'Tuition quotation ready',
                                'Your tuition quotation is ready. Please see the Finance office. '
                                'After this, the Registrar will give your Certificate of Enrolment.',
                                details)
        return existing is None

from db import query


class Finance:
    """Mga datos para sa Finance: ang pila ng mga applicant na handa nang i-quote."""

    # Hindi kasama ang mga applicant ng term na naka-archive na
    SCOPE = "a.status = 'For Enrollment' AND (t.admission_status IS NULL OR t.admission_status <> 'Archived')"
    FROM = ("FROM applications a JOIN users u ON u.id = a.user_id "
            "LEFT JOIN programs p ON p.id = a.program_id "
            "LEFT JOIN academic_terms t ON t.id = a.term_id "
            "LEFT JOIN quotations qt ON qt.application_id = a.id")

    @classmethod
    def _filters(cls, q='', state=None):
        """state: None = lahat, 'ready' = wala pang quotation, 'quoted' = may quotation na."""
        where = [cls.SCOPE, 'a.originals_received_at IS NOT NULL']
        params = []
        if state == 'ready':
            where.append('qt.id IS NULL')
        elif state == 'quoted':
            where.append('qt.id IS NOT NULL')
        if q:
            like = f'%{q}%'
            where.append('(u.first_name LIKE %s OR u.last_name LIKE %s OR u.email LIKE %s OR qt.student_id LIKE %s)')
            params += [like, like, like, like]
        return ' AND '.join(where), params

    @classmethod
    def queue_count(cls, q='', state=None):
        where, params = cls._filters(q, state)
        return query(f'SELECT COUNT(*) AS n {cls.FROM} WHERE {where}', params, one=True)['n']

    @classmethod
    def queue(cls, q='', limit=10, offset=0, state=None):
        """Una ang wala pang quotation; sa loob nito, ang pinakamatagal nang naghihintay."""
        where, params = cls._filters(q, state)
        return query(
            'SELECT a.id, a.applicant_type, a.originals_received_at, u.first_name, u.last_name, u.email, '
            'p.code AS program_code, p.name AS program_name, t.term_no, t.academic_year, '
            'qt.id AS quotation_id, qt.student_id, qt.total '
            f'{cls.FROM} WHERE {where} '
            'ORDER BY (qt.id IS NOT NULL) ASC, a.originals_received_at ASC, a.id ASC LIMIT %s OFFSET %s',
            params + [limit, offset])

    @classmethod
    def counts(cls):
        waiting = query(f'SELECT COUNT(*) AS n {cls.FROM} WHERE {cls.SCOPE} AND a.originals_received_at IS NULL',
                        (), one=True)['n']
        return {'ready': cls.queue_count(state='ready'), 'quoted': cls.queue_count(state='quoted'),
                'waiting': waiting}

import os

class Coe:
    """Certificate of Enrolment: in-upload ng Registrar, kinukuha ng student. Dito rin nagiging Enrolled ang applicant."""

    MAX_BYTES = 5 * 1024 * 1024
    TYPES = {
        '.pdf': ('application/pdf', (b'%PDF',)),
        '.jpg': ('image/jpeg', (b'\xff\xd8\xff',)),
        '.jpeg': ('image/jpeg', (b'\xff\xd8\xff',)),
        '.png': ('image/png', (b'\x89PNG\r\n\x1a\n',)),
    }
    NOT_ARCHIVED = "(t.admission_status IS NULL OR t.admission_status <> 'Archived')"

    @staticmethod
    def folder():
        return os.path.join(current_app.root_path, 'uploads', 'coe')

    # ---------- mga listahan ----------
    @classmethod
    def ready(cls):
        """Mga na-quote na ng Finance, naghihintay ng COE."""
        return query(
            "SELECT a.id, a.applicant_type, u.first_name, u.last_name, u.email, p.code AS program_code, "
            "t.term_no, t.academic_year, qt.student_id, qt.total "
            "FROM applications a JOIN quotations qt ON qt.application_id = a.id "
            "JOIN users u ON u.id = a.user_id "
            "LEFT JOIN programs p ON p.id = a.program_id "
            "LEFT JOIN academic_terms t ON t.id = a.term_id "
            f"WHERE a.status = 'For Enrollment' AND {cls.NOT_ARCHIVED} "
            "ORDER BY qt.updated_at ASC, a.id ASC")

    @classmethod
    def released(cls, limit=20):
        """Mga kamakailang na-release na COE."""
        return query(
            "SELECT e.id AS enrollment_id, e.enrolled_at, e.student_type, u.first_name, u.last_name, "
            "u.username AS student_id, t.term_no, t.academic_year "
            "FROM enrollments e JOIN users u ON u.id = e.user_id "
            "JOIN academic_terms t ON t.id = e.term_id "
            f"WHERE e.status = 'Enrolled' AND e.coe_stored IS NOT NULL AND {cls.NOT_ARCHIVED} "
            "ORDER BY e.enrolled_at DESC LIMIT %s", (limit,))

    @classmethod
    def for_student(cls, user_id):
        return query(
            "SELECT e.*, t.term_no, t.academic_year FROM enrollments e "
            "JOIN academic_terms t ON t.id = e.term_id WHERE e.user_id = %s ORDER BY e.term_id DESC", (user_id,))

    # ---------- pagsusuri ----------
    @classmethod
    def application_for_release(cls, application_id):
        """Ang application kung puwede nang bigyan ng COE: For Enrollment, may quotation, hindi archived ang term."""
        from models.application import Application
        application = Application.find(application_id)
        if application is None or application.status != 'For Enrollment' or Quotation.find(application.id) is None:
            return None
        if application.term_id:
            term = query("SELECT admission_status FROM academic_terms WHERE id = %s", (application.term_id,), one=True)
            if term and term['admission_status'] == 'Archived':
                return None
        return application

    @classmethod
    def _read_file(cls, file):
        if file is None or not file.filename:
            raise ValueError('Choose the COE file to upload.')
        original = secure_filename(file.filename) or 'coe'
        ext = os.path.splitext(original)[1].lower()
        if ext not in cls.TYPES:
            raise ValueError('The COE must be a PDF, JPG or PNG file.')
        data = file.read(cls.MAX_BYTES + 1)
        if len(data) > cls.MAX_BYTES:
            raise ValueError('The file is too large (5 MB maximum).')
        mime, magics = cls.TYPES[ext]
        if not data.startswith(magics):
            raise ValueError('The file content does not match its type. Upload the real PDF, JPG or PNG.')
        return secrets.token_hex(16) + ext, original[-150:], mime, data

    # ---------- ang mismong pag-release ----------
    @classmethod
    def release(cls, application, file):
        """Itatabi ang COE, gagawing Enrolled ang application at Student ang account."""
        if application.status != 'For Enrollment':
            raise ValueError('This application is not waiting for a COE.')
        quotation = Quotation.find(application.id)
        if quotation is None:
            raise ValueError('The Finance quotation is not ready yet.')
        if not application.birthdate:
            raise ValueError('The applicant has no birthdate on file (needed for the initial password).')
        enrollment = query("SELECT * FROM enrollments WHERE user_id = %s AND term_id = %s",
                           (application.user_id, application.term_id), one=True)
        if enrollment is None:
            raise ValueError('No enrollment record found. Ask Finance to save the quotation again.')

        stored, original, mime, data = cls._read_file(file)
        os.makedirs(cls.folder(), exist_ok=True)
        path = os.path.join(cls.folder(), stored)
        with open(path, 'wb') as handle:
            handle.write(data)

        try:
            execute("UPDATE enrollments SET coe_stored = %s, coe_name = %s, coe_mime = %s, "
                    "coe_uploaded_at = NOW(), status = 'Enrolled', enrolled_at = NOW() WHERE id = %s",
                    (stored, original, mime, enrollment['id']))
            execute("UPDATE applications SET status = 'Enrolled' WHERE id = %s", (application.id,))
            user = User.find(application.user_id)
            if user.role == 'applicant':
                user.promote_to_student(quotation['student_id'], application.birthdate.strftime('%y/%m/%d'))
                user.activate()                          # Student ID at birthday ang gagamitin niya, walang activation link
        except Exception:
            os.remove(path)
            raise

        Notification.notify(
            user, 'You are now enrolled at SIT', 'Certificate of Enrolment ready',
            'Your Certificate of Enrolment is ready and you are now enrolled. Log in to the Student Portal '
            'with your Student ID to see it. Please change your password after your first login.',
            [('Student ID', quotation['student_id']),
             ('Initial password', 'Your birthday written as YY/MM/DD'),
             ('Student Portal', url_for('portal.login', _external=True))])

    # ---------- pagkuha ng file ----------
    @staticmethod
    def file_row(enrollment_id, user_id=None):
        sql, params = "SELECT * FROM enrollments WHERE id = %s AND coe_stored IS NOT NULL", [enrollment_id]
        if user_id is not None:
            sql += " AND user_id = %s"
            params.append(user_id)
        return query(sql, params, one=True)

    @classmethod
    def send(cls, row):
        if row is None:
            abort(404)
        path = os.path.join(cls.folder(), row['coe_stored'])
        if not os.path.isfile(path):
            abort(404)
        response = send_file(path, mimetype=row['coe_mime'], download_name=row['coe_name'])
        response.headers['X-Content-Type-Options'] = 'nosniff'
        return response

STAFF_ROLES = ('registrar', 'finance', 'faculty')
FILTER_ROLES = ('applicant', 'student') + STAFF_ROLES
STUDENT_ID_RE = re.compile(r'^\d{2}-[1-3]-\d{4}$')      # YY-trimester-NNNN, hal. 23-1-0096
PER_PAGE = 10


def person_errors(first, middle, last, email):
    errors = []
    if not (2 <= len(first) <= 50 and NAME_RE.match(first)):
        errors.append('First name must be 2 to 50 letters.')
    if middle and not (len(middle) <= 50 and NAME_RE.match(middle)):
        errors.append('Middle name may only contain letters, up to 50 characters.')
    if not (2 <= len(last) <= 50 and NAME_RE.match(last)):
        errors.append('Last name must be 2 to 50 letters.')
    if not EMAIL_RE.match(email) or len(email) > 120:
        errors.append('Please enter a valid email address.')
    return errors


def send_activation(user):

@admin_users_bp.route('/new', methods=['GET', 'POST'])
@roles_required('admin')
def create():
    today = date.today().isoformat()
    if request.method == 'GET':
        return render_template('admin/user_form.html', mode='create', form={}, today=today)

    f = request.form
    role = f.get('role', '')
    first = clean_name(f.get('first_name', ''))
    middle = clean_name(f.get('middle_name', ''))
    last = clean_name(f.get('last_name', ''))
    email = f.get('email', '').strip().lower()
    student_id = f.get('student_id', '').strip()
    birthdate = f.get('birthdate', '').strip()

    errors = person_errors(first, middle, last, email)
    if role not in STAFF_ROLES + ('student',):
        errors.append('Choose a valid role.')

    existing = User.find_by_email(email) if EMAIL_RE.match(email) else None
    initial_password = None
    if role == 'student':
        if not STUDENT_ID_RE.match(student_id):
            errors.append('Student ID must look like 23-1-0096 (year, trimester 1-3, number).')
        elif User.find_by_username(student_id):
            errors.append('That Student ID is already assigned to another account.')
        try:
            born = datetime.strptime(birthdate, '%Y-%m-%d').date()
            if born >= date.today():
                raise ValueError
            initial_password = born.strftime('%y/%m/%d')       # YY/MM/DD
        except ValueError:
            errors.append('Enter a valid birth date.')
        if existing and existing.role != 'applicant':
            errors.append('That email already belongs to another account.')
    elif existing:
        errors.append('That email already belongs to another account.')

    if errors:
        for message in errors:
            flash(message, 'danger')
        return render_template('admin/user_form.html', mode='create', form=f, today=today)

    if role == 'student' and existing:                          # applicant -> student
        existing.update_details(first, middle or None, last, email)
        existing.promote_to_student(student_id, initial_password)
        user = User.find_by_email(email)
        AuditLog.record('Applicant promoted to student', f'{email} -> {student_id}')
    else:
        user = User(first, last, email, role, middle or None)
        if role == 'student':
            user.username = student_id
            user.set_password(initial_password)
        else:
            user.set_password(secrets.token_urlsafe(24))       # pansamantala; sila ang pipili sa activation
        user.is_activated = 0
        user.save()
        user = User.find_by_email(email)
        AuditLog.record('Account created', f'{role}: {student_id or email}')

    if send_activation(user):
        flash(f'Account created. An activation link was sent to {email}.', 'success')
    else:
        flash('Account created, but the email could not be sent. Use "Resend activation link" later.', 'warning')
    return redirect(url_for('admin_users.index'))
